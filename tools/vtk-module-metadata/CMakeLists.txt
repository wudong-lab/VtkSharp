cmake_minimum_required(VERSION 3.25)
project(VtkSharpModuleMetadata LANGUAGES CXX)

if(NOT DEFINED VTK_DIR)
  message(FATAL_ERROR "Pass -DVTK_DIR=<matching VTK CMake package directory>.")
endif()
if(NOT DEFINED VTKSHARP_CONFIGURATION)
  set(VTKSHARP_CONFIGURATION Release)
endif()

include("${CMAKE_CURRENT_LIST_DIR}/../../src/bindings/VtkSharp.Native/vtksharp.modules.generated.cmake")
find_package(VTK CONFIG REQUIRED COMPONENTS ${VTKSHARP_VTK_COMPONENTS})

function(vtksharp_json_escape output value)
  string(REPLACE "\\" "\\\\" escaped "${value}")
  string(REPLACE "\"" "\\\"" escaped "${escaped}")
  set(${output} "${escaped}" PARENT_SCOPE)
endfunction()

function(vtksharp_json_array output)
  set(json "[")
  set(first TRUE)
  foreach(value IN LISTS ARGN)
    vtksharp_json_escape(escaped "${value}")
    if(NOT first)
      string(APPEND json ",")
    endif()
    string(APPEND json "\"${escaped}\"")
    set(first FALSE)
  endforeach()
  string(APPEND json "]")
  set(${output} "${json}" PARENT_SCOPE)
endfunction()

if(DEFINED VTKSHARP_METADATA_OUTPUT)
  set(metadata_output "${VTKSHARP_METADATA_OUTPUT}")
else()
  set(metadata_output "${CMAKE_CURRENT_LIST_DIR}/../../src/generator/config/vtksharp.native-metadata.json")
endif()
get_filename_component(metadata_output "${metadata_output}" ABSOLUTE)
get_filename_component(metadata_directory "${metadata_output}" DIRECTORY)
file(MAKE_DIRECTORY "${metadata_directory}")

set(metadata "{\n  \"schemaVersion\": 1,\n  \"vtkVersion\": \"${VTK_VERSION}\",\n  \"platform\": \"windows-x64\",\n  \"configuration\": \"${VTKSHARP_CONFIGURATION}\",\n  \"modules\": [\n")
list(LENGTH VTKSHARP_VTK_COMPONENTS component_count)
set(component_index 0)
foreach(component IN LISTS VTKSHARP_VTK_COMPONENTS)
  math(EXPR component_index "${component_index} + 1")
  set(target "VTK::${component}")
  foreach(property IN ITEMS depends private_depends optional_depends implements implementable library_name needs_autoinit)
    get_target_property(${property}_value "${target}" "INTERFACE_vtk_module_${property}")
    if(${property}_value STREQUAL "${property}_value-NOTFOUND")
      set(${property}_value "")
    endif()
    list(SORT ${property}_value)
  endforeach()
  vtksharp_json_array(depends_json ${depends_value})
  vtksharp_json_array(private_depends_json ${private_depends_value})
  vtksharp_json_array(optional_depends_json ${optional_depends_value})
  vtksharp_json_array(implements_json ${implements_value})
  vtksharp_json_escape(module_name "vtk${component}")
  vtksharp_json_escape(library_name "${library_name_value}")
  if(implementable_value)
    set(implementable_json true)
  else()
    set(implementable_json false)
  endif()
  if(needs_autoinit_value)
    set(needs_autoinit_json true)
  else()
    set(needs_autoinit_json false)
  endif()
  string(APPEND metadata "    {\"module\": \"${module_name}\", \"depends\": ${depends_json}, \"privateDepends\": ${private_depends_json}, \"optionalDepends\": ${optional_depends_json}, \"implements\": ${implements_json}, \"implementable\": ${implementable_json}, \"libraryName\": \"${library_name}\", \"needsAutoinit\": ${needs_autoinit_json}}")
  if(component_index LESS component_count)
    string(APPEND metadata ",")
  endif()
  string(APPEND metadata "\n")
endforeach()
string(APPEND metadata "  ]\n}\n")
file(WRITE "${metadata_output}" "${metadata}")
message(STATUS "Wrote VTK module metadata snapshot: ${metadata_output}")
